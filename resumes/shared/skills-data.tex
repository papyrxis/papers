\def\addtocatlist#1#2{%
  \ifcsdef{Skill#1}{%
    \expandafter\appto\csname Skill#1\endcsname{, #2}%
  }{%
    \expandafter\gdef\csname Skill#1\endcsname{#2}%
  }%
}

\newcommand{\newskill}[3]{%
