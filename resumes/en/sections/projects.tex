\newcommand{\projcdin}{%
  \proj{CDIN}{C11 $\cdot$ Lua 5.4 $\cdot$ SDL3}{%
    \item I created CDIN, a project to build a modular text editor. Its core stores document content and edit history, the Lua-based runtime executes commands and plugins, and a separate layer handles operating-system services. The terminal interface is the primary experience; the SDL-based graphical interface is still experimental.%
  }{cdin-editor/cdin}%
}

\newcommand{\projgland}{%
  \proj{GlandJS}{TypeScript $\cdot$ Node.js $\cdot$ Event-driven}{%
    \item GlandJS is a framework for building server-side applications with TypeScript and Node.js. Instead of calling one another directly, parts of an application communicate through messages and events. The framework manages service creation and keeps the core separate from the web server, so the request-handling layer can change without rewriting the main application logic.%
  }{glandjs/gland}%
}

\newcommand{\projinfra}{%
  \proj{Developer Tools}{Go $\cdot$ Python $\cdot$ C11}{%
    \item A set of tools for everyday software development: Pipeline defines build, test, and release tasks once and runs them locally or in GitHub Actions and GitLab CI; gix simplifies Git branch and version management; psx checks whether a repository has the files and structure required by its project rules; and log.c is a lightweight logging library for C programs.%
  }{m-mdy-m/pipeline}%
}

\newcommand{\projftx}{%
  \proj{FTX.vim}{Vim Script $\cdot$ Vim $\cdot$ Neovim}{%
    \item A file-tree plugin for Vim and Neovim. It updates the tree asynchronously to keep the editor responsive and displays Git status alongside files and directories.%
  }{m-mdy-m/ftx.vim}%
}
