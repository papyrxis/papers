\newcommand{\projcdin}{%
  \proj{CDIN}{C11 $\cdot$ Lua 5.4 $\cdot$ SDL3}{%
    \item I work on the architecture and integration of a modular, cross-platform text editor. The Kernel owns document state and undo/redo; the Lua Runtime manages commands and plugins; the Backend provides sessions and OS services; and Frontends provide TUI and GUI-SDL interfaces. The GUI-SDL renderer remains experimental.%
  }{cdin-editor/cdin}%
}

\newcommand{\projgland}{%
  \proj{GlandJS}{TypeScript $\cdot$ Node.js $\cdot$ Event-driven}{%
    \item A backend framework for modular Node.js services. Components communicate through a message bus; dependency injection and module/controller metadata handle composition; and HTTP adapters stay separate from the core. This keeps application logic from depending directly on one server or transport.%
  }{glandjs/gland}%
}
