\newcommand{\projgland}{%
  \proj{GlandJS}{TypeScript · Node.js · Event-driven}{%
    \item Creator and primary maintainer of an event-driven Node.js framework. Its package ecosystem combines module bootstrapping, dependency injection, lifecycle hooks, event messaging, and a transport-independent HTTP layer with adapters for Express, Fastify, Koa, Hono, and node:http. Core packages are published under the \texttt{@glandjs} npm scope.%
  }{glandjs/gland}%
}

\newcommand{\projcdin}{%
  \proj{CDIN Editor}{C11 · Lua 5.4 · Protocol design}{%
    \item Creator and architect of a modular text-editor project. Its independently maintained components separate binary-safe document state (Kernel), Lua-based editor behavior (Runtime), OS/session services (Backend), and protocol-driven presentation (Frontend). Product composition and the first interactive release remain in active development; the SDL desktop renderer is experimental.%
  }{cdin-editor/cdin}%
}
