\chapter*{فهرست نمادها و اختصارات}
\addcontentsline{toc}{chapter}{فهرست نمادها و اختصارات}
\markboth{فهرست نمادها و اختصارات}{فهرست نمادها و اختصارات}

این فهرست برای مراجعه‌ی سریع به نمادها و اختصارهایی است که در سه فصل
کتاب به کار رفته‌اند. یک نماد ممکن است در دو حوزه معنای متفاوتی داشته
باشد؛ در چنین مواردی، زمینه‌ی رابطه تعیین می‌کند کدام معنا منظور است.

\section*{نمادهای ریاضی و نظریه‌ی اطلاعات}
\begin{description}
  \item[\(x\)] یک رخداد یا نتیجه‌ی مشخص از مجموعه‌ی حالت‌های ممکن.
  \item[\(X,Y\)] متغیرهای تصادفی؛ در مدل ارتباط، معمولاً \(X\) ورودی و \(Y\) خروجی کانال است.
  \item[\(p(x)\)] احتمال رخداد یا نتیجه‌ی \(x\).
  \item[\(I(x)\)] خوداطلاعاتیِ رخداد \(x\).
  \item[\(H(X)\)] آنتروپی شانونِ متغیر تصادفی \(X\).
  \item[\(H(X\mid Y)\)] آنتروپی شرطی؛ عدم‌قطعیت باقی‌مانده درباره‌ی \(X\) پس از دانستن \(Y\).
  \item[\(I(X;Y)\)] اطلاعات متقابل میان \(X\) و \(Y\).
  \item[\(M\)] تعداد پیام‌ها یا رشته‌های متمایز در مدل شمارش حالت‌ها.
  \item[\(s\)] اندازه‌ی الفبای نمادها در یک منبع یا کد.
  \item[\(n\)] تعداد جایگاه‌های نماد در یک رشته.
  \item[\(C\)] ظرفیت کانال در رابطه‌ی شانون--هارتلی؛ در مدارها می‌تواند نماد ظرفیت خازن باشد.
  \item[\(B\)] پهنای باند کانال، بر حسب هرتز.
  \item[\(S/N\)] نسبت توان سیگنال به توان نویز.
  \item[\(k_{\mathrm B}\)] ثابت بولتزمن.
  \item[\(T\)] دمای مطلق در رابطه‌ی لاندائر.
\end{description}

\section*{نمادهای سیگنال و الکتریسیته}
\begin{description}
  \item[\(f_s\)] نرخ نمونه‌برداری.
  \item[\(T_s\)] فاصله‌ی زمانی میان دو نمونه‌ی متوالی.
  \item[\(f_{\max}\)] بیشینه‌ی فرکانس مؤلفه‌های سیگنال محدودباند.
  \item[\(q\)] بار الکتریکی؛ مقدار آن می‌تواند مثبت یا منفی باشد.
  \item[\(e\)] بار بنیادی، یعنی قدرمطلق بار الکتریکی پروتون یا الکترون.
  \item[\(\vec E\)] میدان الکتریکی.
  \item[\(\vec F\)] نیروی الکتریکی.
  \item[\(U,\Delta U\)] انرژی پتانسیل الکتریکی و تغییر آن.
  \item[\(V,\Delta V\)] پتانسیل الکتریکی و اختلاف پتانسیل میان دو نقطه.
  \item[\(I\)] جریان الکتریکی در روابط مدار؛ در نظریه‌ی اطلاعات، حرف \(I\) در \(I(x)\) یا \(I(X;Y)\) معنای دیگری دارد.
  \item[\(Q,\Delta Q\)] بار الکتریکی منتقل‌شده یا تغییر بار.
  \item[\(R\)] مقاومت الکتریکی.
  \item[\(P,p(t)\)] توان الکتریکی؛ \(p(t)\) توان لحظه‌ای است.
  \item[\(C\)] ظرفیت خازن در روابط مدار.
  \item[\(L\)] اندوکتانس یا ضریب القای سلف.
  \item[\(v(t),i(t)\)] ولتاژ و جریان لحظه‌ای.
\end{description}

\section*{اختصارات}
\begin{description}
  \item[\lr{DIKW}] سرواژه‌ی \lr{Data, Information, Knowledge, Wisdom}؛ داده، اطلاعات، دانش و خرد.
  \item[\lr{DNA}] \lr{Deoxyribonucleic Acid}؛ دئوکسی‌ریبونوکلئیک اسید، مولکول حامل اطلاعات وراثتی.
  \item[\lr{ADC}] \lr{Analog-to-Digital Converter}؛ مبدل آنالوگ به دیجیتال.
  \item[\lr{AC}] \lr{Alternating Current}؛ جریان متناوب.
  \item[\lr{DC}] \lr{Direct Current}؛ جریان مستقیم.
  \item[\lr{SI}] \lr{International System of Units}؛ دستگاه بین‌المللی یکاها.
  \item[\lr{MTL}] \lr{Medial Temporal Lobe}؛ لوب گیجگاهی میانی.
\end{description}
